# syntax=docker/dockerfile:1

# Build stage: resolve dependencies first (cached while pom.xml is unchanged), then package.
FROM maven:3-eclipse-temurin-25 AS build
WORKDIR /src
COPY pom.xml .
RUN mvn -B -q dependency:go-offline
COPY src ./src
RUN mvn -B -q package -DskipTests

# Runtime stage: only the JRE and the application jar, run as an unprivileged user.
FROM eclipse-temurin:25-jre
WORKDIR /app
RUN useradd --system --uid 10001 app
# The jar Maven builds: set <finalName>app</finalName> in pom.xml, or change the name here.
COPY --from=build /src/target/app.jar app.jar
USER app
EXPOSE 8080
ENTRYPOINT ["java", "-jar", "/app/app.jar"]
